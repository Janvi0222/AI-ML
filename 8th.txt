father(X, Y) :-
    male(X),
    parent(X, Y).

mother(X, Y) :-
    female(X),
    parent(X, Y).

grandfather(X, Y) :-
    male(X),
    parent(X, Z),
    parent(Z, Y).

grandmother(X, Y) :-
    female(X),
    parent(X, Z),
    parent(Z, Y).

brother(X, Y) :-
    male(X),
    parent(P, X),
    parent(P, Y),
    X \= Y.

sister(X, Y) :-
    female(X),
    parent(P, X),
    parent(P, Y),
    X \= Y.

uncle(X, Y) :-
    male(X),
    parent(P, Y),
    brother(X, P).

aunt(X, Y) :-
    female(X),
    parent(P, Y),
    sister(X, P).

nephew(X, Y) :-
    male(X),
    parent(P, X),
    (brother(P, Y) ; sister(P, Y)).

niece(X, Y) :-
    female(X),
    parent(P, X),
    (brother(P, Y) ; sister(P, Y)).

cousin(X, Y) :-
    parent(P1, X),
    parent(P2, Y),
    (brother(P1, P2) ; sister(P1, P2)).
